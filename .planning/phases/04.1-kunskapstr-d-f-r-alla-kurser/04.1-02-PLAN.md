const COURSES = ["B", "C", "D"];
const states = {};

function loadState(course) {
  try { return JSON.parse(localStorage.getItem(`sfi-checklist-${course}`) || "{}"); } catch { return {}; }
}
function saveState(course, st) {
  try { localStorage.setItem(`sfi-checklist-${course}`, JSON.stringify(st)); } catch {}
}

// Load all courses on page init
COURSES.forEach(c => { states[c] = loadState(c); });

**2. Tab switching (per D-01, D-02)**

```javascript
let activeCourse = "B"; // D-02: default to Kurs B

const treeTabs = document.querySelectorAll('.tree-tab');
const treePanels = document.querySelectorAll('.course-tree-panel');
const headerSubtitle = document.getElementById('tree-header-subtitle');

const tabActiveClasses = {
  B: ["border-cyan-600", "text-cyan-700"],
  C: ["border-violet-600", "text-violet-700"],
  D: ["border-rose-600", "text-rose-700"],
};
const tabInactiveClasses = ["border-transparent", "text-gray-400"];

function switchTreeTab(course) {
  activeCourse = course;

  // Toggle panel visibility
  treePanels.forEach(p => p.classList.toggle('hidden', p.id !== `tree-panel-${course}`));

  // Update tab styles
  treeTabs.forEach(t => {
    const tc = t.dataset.course;
    // Remove all active classes
    Object.values(tabActiveClasses).forEach(cls => cls.forEach(c => t.classList.remove(c)));
    tabInactiveClasses.forEach(c => t.classList.remove(c));

    if (tc === course) {
      tabActiveClasses[course].forEach(c => t.classList.add(c));
      t.setAttribute("aria-selected", "true");
    } else {
      tabInactiveClasses.forEach(c => t.classList.add(c));
      t.setAttribute("aria-selected", "false");
    }
  });

  // Update header subtitle
  if (headerSubtitle) {
    headerSubtitle.textContent = `Kurs ${course} \u00B7 Dina formagar vaxer`;
  }
}

treeTabs.forEach(t => {
  t.addEventListener('click', () => switchTreeTab(t.dataset.course));
});
```

**4. Click/keyboard handling — scoped to course**

```javascript
document.querySelectorAll('.leaf-node').forEach(leaf => {
  function toggle() {
    const id = leaf.dataset.goalId;
    const skill = leaf.dataset.skill;
    const course = leaf.dataset.course;
    const wasChecked = leaf.getAttribute('data-checked') === 'true';
    const nowChecked = !wasChecked;

    // Update state
    if (nowChecked) { states[course][id] = true; }
    else { delete states[course][id]; }
    saveState(course, states[course]);

    // Update leaf visual
    leaf.setAttribute('data-checked', String(nowChecked));
    leaf.setAttribute('aria-checked', String(nowChecked));

    // Pop animation
    leaf.classList.remove('leaf-pop');
    void leaf.offsetWidth;
    leaf.classList.add('leaf-pop');

    // Number color — scope query to course panel
    const panel = document.getElementById(`tree-panel-${course}`);
    const num = panel.querySelector(`[data-num-for="${id}"]`);
    if (num) num.setAttribute('fill', nowChecked ? 'white' : '');

    // Branch
    const branch = panel.querySelector(`[data-branch-for="${id}"]`);
    if (branch) branch.setAttribute('data-checked', String(nowChecked));

    // Show goal text — course-scoped panel
    showGoalText(course, skill, id, nowChecked);

    // Update progress
    updateProgress(course, skill);

    // Update crown glow
    updateCrown(course, skill);

    // Update radar for this course
    updateRadar(course);
  }

  leaf.addEventListener('click', toggle);
  leaf.addEventListener('keydown', (e) => {
    if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); toggle(); }
  });
});
```

function updateProgress(course, skillId) {
  const skills = allSkillsClient[course];
  if (!skills) return;
  const skill = skills.find(s => s.id === skillId);
  if (!skill) return;
  const checked = skill.goalIds.filter(id => states[course][id]).length;
  const total = skill.goalIds.length;
  const pct = Math.round((checked / total) * 100);

  const progressEl = document.querySelector(`[data-tree-progress="${course}-${skillId}"]`);
  if (progressEl) progressEl.textContent = `${checked} av ${total}`;

  const bar = document.querySelector(`[data-progress-bar="${course}-${skillId}"]`);
  if (bar) bar.style.width = `${pct}%`;
}

**8. Radar chart — per course (per D-08, D-09)**

```javascript
function updateRadar(course) {
  const maxR = 110;
  const cx = 150, cy = 150;
  const skills = allSkillsClient[course];
  if (!skills) return;

  const pcts = skills.map(s => {
    const checked = s.goalIds.filter(id => states[course][id]).length;
    return checked / s.goalIds.length;
  });

  const pts = pcts.map((p, i) => {
    const angleDeg = -90 + i * 60;
    const rad = (angleDeg * Math.PI) / 180;
    const r = maxR * p;
    return `${Math.round(cx + r * Math.cos(rad))},${Math.round(cy + r * Math.sin(rad))}`;
  });

  const polygon = document.getElementById(`radar-polygon-${course}`);
  if (polygon) polygon.setAttribute('points', pts.join(' '));

  skills.forEach((s, i) => {
    const angleDeg = -90 + i * 60;
    const rad = (angleDeg * Math.PI) / 180;
    const r = maxR * pcts[i];
    const dot = document.getElementById(`radar-dot-${s.id}-${course}`);
    if (dot) {
      dot.setAttribute('cx', String(Math.round(cx + r * Math.cos(rad))));
      dot.setAttribute('cy', String(Math.round(cy + r * Math.sin(rad))));
    }
    const el = document.getElementById(`pct-${s.id}-${course}`);
    if (el) el.textContent = `${Math.round(pcts[i] * 100)} %`;
  });
}
```
