**Step 4: Parameterize `computePositions`**

Modify the existing function to accept a tier parameter:

```typescript
function computePositions(n: number, tier: typeof TIERS.medium) {
  const cx = 150, cy = 170;
  const positions: { x: number; y: number }[] = [];

  if (n <= 6) {
    const r = tier.singleR;
    const arcStart = tier.singleArcStart, arcEnd = tier.singleArcEnd;
    for (let i = 0; i < n; i++) {
      const angle = n === 1 ? 270 : arcStart + ((arcEnd - arcStart) / (n - 1)) * i;
      const rad = (angle * Math.PI) / 180;
      positions.push({ x: Math.round(cx + r * Math.cos(rad)), y: Math.round(cy + r * Math.sin(rad)) });
    }
  } else {
    const inner = Math.floor(n / 2);
    const outer = n - inner;
    const r1 = tier.innerOrbit, s1Start = tier.innerArcStart, s1End = tier.innerArcEnd;
    for (let i = 0; i < inner; i++) {
      const angle = inner === 1 ? 270 : s1Start + ((s1End - s1Start) / (inner - 1)) * i;
      const rad = (angle * Math.PI) / 180;
      positions.push({ x: Math.round(cx + r1 * Math.cos(rad)), y: Math.round(cy + r1 * Math.sin(rad)) });
    }
    const r2 = tier.outerOrbit, s2Start = tier.outerArcStart, s2End = tier.outerArcEnd;
    for (let i = 0; i < outer; i++) {
      const angle = outer === 1 ? 270 : s2Start + ((s2End - s2Start) / (outer - 1)) * i;
      const rad = (angle * Math.PI) / 180;
      positions.push({ x: Math.round(cx + r2 * Math.cos(rad)), y: Math.round(cy + r2 * Math.sin(rad)) });
    }
  }
  return positions;
}
```

const allSkills: Record<string, any[]> = {};
