c) Add Min plan localStorage functions (per D-06). Place after existing saveState function:
```javascript
const PLAN_STORAGE_KEY = 'sfi-minplan';

function loadPlanState() {
  try {
    return JSON.parse(localStorage.getItem(PLAN_STORAGE_KEY) || '{"checked":{},"custom":{}}');
  } catch {
    return { checked: {}, custom: {} };
  }
}

function savePlanState(state) {
  try {
    localStorage.setItem(PLAN_STORAGE_KEY, JSON.stringify(state));
  } catch {}
}
```

d) Add checkbox handler for plan items. Use delegated event handling on `#panel-plan` for both pre-filled checkboxes (`[data-plan-id]`) and custom checkboxes (`[data-plan-custom]`):
```javascript
const planPanel = document.getElementById('panel-plan');
if (planPanel) {
  planPanel.addEventListener('change', (e) => {
    const cb = e.target.closest('input[type="checkbox"]');
    if (!cb) return;
    const state = loadPlanState();
    const planId = cb.dataset.planId;
    const customId = cb.dataset.planCustom;
    const key = planId || customId;
    if (key) {
      if (cb.checked) { state.checked[key] = true; }
      else { delete state.checked[key]; }
      savePlanState(state);
    }
  });

  // Text input persistence (per D-04) — save on input event
  planPanel.addEventListener('input', (e) => {
    const input = e.target.closest('input[type="text"][data-plan-text]');
    if (!input) return;
    const state = loadPlanState();
    state.custom[input.dataset.planText] = input.value;
    savePlanState(state);
  });
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| localStorage read/write | User-controlled storage — data could be manually modified |
| Custom text inputs | Free-form user text rendered in DOM |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | localStorage sfi-minplan | accept | Low risk — student's own device, no auth, no server sync. Tampered data only affects their own view. JSON.parse wrapped in try/catch with safe default. |
| T-03-02 | Injection | Custom text input -> DOM | mitigate | Custom text inputs are `<input type="text">` elements. Values restored via `.value =` property assignment (not innerHTML), so no XSS vector. If any display renders custom text, use textContent not innerHTML. |
| T-03-03 | Information Disclosure | Print output | accept | Print shows student's checked items. This is intentional — the teacher prints it for classroom use. No sensitive data beyond study preferences. |
