**1. Tab Switching (D-03, D-04, D-05, D-06)**
```javascript
const tabs = document.querySelectorAll('.course-tab');
const panels = document.querySelectorAll('.course-panel');

function switchTab(course) {
  panels.forEach(p => {
    p.classList.toggle('hidden', p.id !== `panel-${course}`);
  });
  tabs.forEach(t => {
    const isActive = t.dataset.course === course;
    t.classList.toggle('border-blue-500', isActive);
    t.classList.toggle('text-gray-900', isActive);
    t.classList.toggle('border-transparent', !isActive);
    t.classList.toggle('text-gray-400', !isActive);
  });
  // Per D-04: scroll to top of content area on tab switch
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

tabs.forEach(tab => {
  tab.addEventListener('click', () => switchTab(tab.dataset.course));
});
```
Per D-06: Do NOT save active tab to localStorage.

**2. localStorage Persistence (D-14, INTER-02)**
Per Pitfall 5: Wrap ALL localStorage calls in try/catch.
```javascript
const STORAGE_KEY = (course) => `sfi-checklist-${course}`;

function loadState(course) {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY(course))) || {};
  } catch {
    return {};
  }
}

function saveState(course, state) {
  try {
    localStorage.setItem(STORAGE_KEY(course), JSON.stringify(state));
  } catch {
    // Storage full or disabled -- silently degrade
  }
}
```

**3. Checkbox Handling + Save Indicator (D-12, INTER-01, INTER-04)**
```javascript
document.querySelectorAll('input[type="checkbox"][data-goal-id]').forEach(cb => {
  cb.addEventListener('change', () => {
    const course = cb.dataset.course;
    const goalId = cb.dataset.goalId;
    const state = loadState(course);
    state[goalId] = cb.checked;
    if (!cb.checked) delete state[goalId]; // Clean up unchecked
    saveState(course, state);
    
    // Show "Sparat!" indicator (D-12)
    showSavedIndicator(goalId);
    
    // Update progress count (D-13)
    updateProgressCounts(course);
  });
});

function showSavedIndicator(goalId) {
  const indicator = document.querySelector(`[data-saved-for="${goalId}"]`);
  if (!indicator) return;
  // Reset animation by cloning
  indicator.classList.remove('hidden');
  const clone = indicator.cloneNode(true);
  indicator.parentNode.replaceChild(clone, indicator);
  // Hide after animation completes (1.5s)
  setTimeout(() => clone.classList.add('hidden'), 1500);
}
```

**4. Progress Counts (D-13)**
```javascript
function updateProgressCounts(course) {
  const panel = document.getElementById(`panel-${course}`);
  if (!panel) return;
  const progressElements = panel.querySelectorAll('[data-progress]');
  progressElements.forEach(el => {
    const key = el.dataset.progress; // e.g., "B-horforstaelse"
    const domainKey = key.split('-').slice(1).join('-'); // Extract domain part
    // Count checked checkboxes in this domain section
    const section = el.closest('section');
    if (!section) return;
    const total = section.querySelectorAll('input[type="checkbox"]').length;
    const checked = section.querySelectorAll('input[type="checkbox"]:checked').length;
    el.textContent = `${checked} av ${total} avklarade`;
  });
}
```

**5. Reset Flow (D-15, D-16, D-17, INTER-05)**
```javascript
['B', 'C', 'D'].forEach(course => {
  const resetBtn = document.querySelector(`[data-reset="${course}"]`);
  const confirmDiv = document.querySelector(`[data-confirm="${course}"]`);
  if (!resetBtn || !confirmDiv) return;

  resetBtn.addEventListener('click', () => {
    resetBtn.classList.add('hidden');
    confirmDiv.classList.remove('hidden');
  });

  confirmDiv.querySelector('.cancel')?.addEventListener('click', () => {
    confirmDiv.classList.add('hidden');
    resetBtn.classList.remove('hidden');
  });

  confirmDiv.querySelector('.confirm')?.addEventListener('click', () => {
    // Clear localStorage for this course (D-17: per-course only)
    try { localStorage.removeItem(STORAGE_KEY(course)); } catch {}
    // Uncheck all checkboxes in this panel
    document.querySelectorAll(`#panel-${course} input[type="checkbox"]`)
      .forEach(cb => { cb.checked = false; });
    // Update all progress counts
    updateProgressCounts(course);
    // Hide confirmation, show reset button
    confirmDiv.classList.add('hidden');
    resetBtn.classList.remove('hidden');
  });
});
```

